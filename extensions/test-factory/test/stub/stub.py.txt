"""A toggleable dependency stub for test-factory scenarios.

Routes:
  GET /ctl?mode=accept|reject|delay|down  -> set the mode (returns the new mode)
  GET /health                             -> 200 while accepting, 503 when rejecting
  GET /...                                -> accept: 200; reject: 503; delay: hang;
                                             down: hang; every response carries an
                                             `X-Stub-Mode` header.

`/ctl` always answers 200 so a test can switch modes even from `reject`.
"""
import os
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
from urllib.parse import urlparse, parse_qs

MODE_FILE = "/tmp/stub-mode"


def mode() -> str:
    try:
        with open(MODE_FILE) as fh:
            return fh.read().strip() or "accept"
    except OSError:
        return "accept"


def set_mode(value: str) -> None:
    with open(MODE_FILE, "w") as fh:
        fh.write(value)


class Handler(BaseHTTPRequestHandler):
    protocol_version = "HTTP/1.1"

    def log_message(self, *args):  # quiet
        pass

    def do_GET(self):
        path = urlparse(self.path).path
        if path == "/ctl":
            value = (parse_qs(urlparse(self.path).query).get("mode") or ["accept"])[0]
            set_mode(value)
            self._reply(200, value.encode(), value)
            return
        current = mode()
        if current in ("down", "delay"):
            # Never answer: model an unreachable or non-responding dependency.
            import time
            time.sleep(3600)
            return
        if path == "/health":
            code = 200 if current == "accept" else 503
            self._reply(code, b"health", current)
            return
        code = 200 if current == "accept" else 503
        self._reply(code, b'{"stub":"ok"}', current)

    def _reply(self, code: int, body: bytes, m: str) -> None:
        self.send_response(code)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(body)))
        self.send_header("X-Stub-Mode", m)
        self.end_headers()
        self.wfile.write(body)


if __name__ == "__main__":
    ThreadingHTTPServer(("0.0.0.0", 8080), Handler).serve_forever()
